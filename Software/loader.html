Como usar

<link rel="stylesheet" href="loader.css">
<script src="loader.js" defer></script>

<!-- opção A: overlay já no HTML (esconde sozinho ao carregar) -->
<div class="rh-loader" data-rh-loader>
  <div class="rh-loader__box">
    <div class="rh-loader__ring">
      <img class="rh-loader__icon" src="assets/rehabit-icon.png" alt="">
    </div>
    <p class="rh-loader__text">Carregando<span class="rh-dots"></span></p>
  </div>
</div>

<!-- opção B: via JS -->
RehabitLoader.show('Carregando');
RehabitLoader.hide();
RehabitLoader.during(fetch('/api/x'), 'Buscando');

<!-- tema escuro: classe rh-loader--dark + rehabit-icon-dark.png -->
<!-- dentro de um card: adicione rh-loader--inline -->

<!-- dentro da página (lista, tabela, menu, janela): o bíceps compacto -->
lista.innerHTML = RehabitLoader.inline('Buscando');
<!-- no meio de um texto (tema escuro pelo body.dark) -->
<p><span class="rh-biceps" aria-hidden="true"></span>Carregando...</p>
